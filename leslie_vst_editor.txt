class RotaryKnob : public juce::Slider
{
public:
    RotaryKnob()
    {
        setSliderStyle(juce::Slider::RotaryHorizontalVerticalDrag);
        setTextBoxStyle(juce::Slider::TextBoxBelow, false, 80, 20);
        setColour(juce::Slider::rotarySliderFillColourId, juce::Colour(0xff4A90E2));
        setColour(juce::Slider::rotarySliderOutlineColourId, juce::Colour(0xff2C3E50));
        setColour(juce::Slider::textBoxTextColourId, juce::Colours::white);
        setColour(juce::Slider::textBoxBackgroundColourId, juce::Colour(0xff34495E));
    }
};

//==============================================================================
// Rotor Visual Component
//==============================================================================
class RotorVisualizer : public juce::Component, public juce::Timer
{
public:
    RotorVisualizer(LeslieRotaryProcessor& p) : processor(p)
    {
        startTimerHz(60); // 60 FPS for smooth animation
    }
    
    ~RotorVisualizer() override
    {
        stopTimer();
    }
    
    void paint(juce::Graphics& g) override
    {
        auto bounds = getLocalBounds().toFloat();
        
        // Background
        g.setColour(juce::Colour(0xff1E1E1E));
        g.fillRoundedRectangle(bounds, 8.0f);
        
        // Border
        g.setColour(juce::Colour(0xff4A90E2));
        g.drawRoundedRectangle(bounds.reduced(1), 8.0f, 2.0f);
        
        // Horn rotor (top)
        auto hornBounds = bounds.removeFromTop(bounds.getHeight() * 0.5f).reduced(20);
        drawRotor(g, hornBounds, hornAngle, juce::Colour(0xff27AE60), "HORN");
        
        // Drum rotor (bottom) 
        auto drumBounds = bounds.reduced(20);
        drawRotor(g, drumBounds, drumAngle, juce::Colour(0xffF39C12), "DRUM");
        
        // Speed indicator
        auto& engine = processor.getRotaryEngine();
        juce::String speedText = "SLOW";
        juce::Colour speedColor = juce::Colour(0xff27AE60);
        
        if (engine.isRamping())
        {
            speedText = "RAMP";
            speedColor = juce::Colour(0xffF39C12);
        }
        else if (engine.getCurrentHornSpeed() > 3.0f)
        {
            speedText = "FAST"; 
            speedColor = juce::Colour(0xffE74C3C);
        }
        else if (engine.getCurrentHornSpeed() < 0.1f)
        {
            speedText = "BRAKE";
            speedColor = juce::Colour(0xff8E44AD);
        }
        
        g.setColour(speedColor);
        g.setFont(juce::Font(12.0f, juce::Font::bold));
        g.drawText(speedText, bounds.removeFromBottom(20), juce::Justification::centred);
    }
    
    void timerCallback() override
    {
        auto& engine = processor.getRotaryEngine();
        hornAngle = engine.getHornAngle();
        drumAngle = engine.getDrumAngle();
        repaint();
    }
    
private:
    void drawRotor(juce::Graphics& g, juce::Rectangle<float> bounds, 
                   float angle, juce::Colour color, const juce::String& label)
    {
        auto center = bounds.getCentre();
        auto radius = juce::jmin(bounds.getWidth(), bounds.getHeight()) * 0.4f;
        
        // Rotor circle
        g.setColour(color.withAlpha(0.3f));
        g.fillEllipse(center.x - radius, center.y - radius, radius * 2, radius * 2);
        
        g.setColour(color);
        g.drawEllipse(center.x - radius, center.y - radius, radius * 2, radius * 2, 2.0f);
        
        // Rotor blades
        for (int i = 0; i < 4; ++i)
        {
            float bladeAngle = angle + (i * juce::MathConstants<float>::pi * 0.5f);
            float x1 = center.x + std::cos(bladeAngle) * radius * 0.3f;
            float y1 = center.y + std::sin(bladeAngle) * radius * 0.3f;
            float x2 = center.x + std::cos(bladeAngle) * radius * 0.8f;
            float y2 = center.y + std::sin(bladeAngle) * radius * 0.8f;
            
            g.setColour(color);
            g.drawLine(x1, y1, x2, y2, 3.0f);
        }
        
        // Center dot
        g.setColour(color);
        g.fillEllipse(center.x - 3, center.y - 3, 6, 6);
        
        // Label
        g.setFont(juce::Font(10.0f, juce::Font::bold));
        g.setColour(juce::Colours::white);
        auto labelBounds = bounds.removeFromBottom(15);
        g.drawText(label, labelBounds, juce::Justification::centred);
    }
    
    LeslieRotaryProcessor& processor;
    float hornAngle = 0.0f;
    float drumAngle = 0.0f;
    
    JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR(RotorVisualizer)
};

//==============================================================================
// Main Editor Component
//==============================================================================
class LeslieRotaryEditor : public juce::AudioProcessorEditor
{
public:
    LeslieRotaryEditor(LeslieRotaryProcessor& p)
        : AudioProcessorEditor(&p), 
          processor(p),
          rotorVisualizer(p)
    {
        // Set up the main layout
        setSize(800, 600);
        setResizable(true, true);
        
        // Initialize components
        setupControls();
        setupLayout();
        
        // Make components visible
        addAndMakeVisible(rotorVisualizer);
        addAndMakeVisible(speedControlGroup);
        addAndMakeVisible(toneControlGroup);
        addAndMakeVisible(outputControlGroup);
        
        // Attach sliders to parameters
        attachSliders();
    }
    
    ~LeslieRotaryEditor() override = default;
    
    void paint(juce::Graphics& g) override
    {
        // Modern dark theme background
        g.fillAll(juce::Colour(0xff2C3E50));
        
        // Header
        g.setColour(juce::Colour(0xff34495E));
        g.fillRect(0, 0, getWidth(), 60);
        
        g.setColour(juce::Colours::white);
        g.setFont(juce::Font(24.0f, juce::Font::bold));
        g.drawText("LESLIE ROTARY", 20, 10, getWidth() - 40, 40, juce::Justification::centred);
        
        // Draw section dividers
        g.setColour(juce::Colour(0xff4A90E2).withAlpha(0.5f));
        g.drawVerticalLine(getWidth() / 3, 70, getHeight() - 10);
        g.drawVerticalLine((getWidth() / 3) * 2, 70, getHeight() - 10);
    }
    
    void resized() override
    {
        auto bounds = getLocalBounds();
        
        // Header space
        bounds.removeFromTop(70);
        
        // Three column layout
        auto leftSection = bounds.removeFromLeft(bounds.getWidth() / 3).reduced(10);
        auto centerSection = bounds.removeFromLeft(bounds.getWidth() / 2).reduced(10);  
        auto rightSection = bounds.reduced(10);
        
        // Left: Speed controls
        speedControlGroup.setBounds(leftSection);
        
        // Center: Rotor visualizer
        rotorVisualizer.setBounds(centerSection.removeFromTop(200));
        toneControlGroup.setBounds(centerSection);
        
        // Right: Output controls
        outputControlGroup.setBounds(rightSection);
    }
    
private:
    void setupControls()
    {
        // Speed Control Group
        speedControlGroup.setText("SPEED CONTROL");
        speedControlGroup.setColour(juce::GroupComponent::outlineColourId, juce::Colour(0xff4A90E2));
        speedControlGroup.setColour(juce::GroupComponent::textColourId, juce::Colours::white);
        
        hornSlowKnob.setTextValueSuffix(" Hz");
        hornFastKnob.setTextValueSuffix(" Hz");
        drumSlowKnob.setTextValueSuffix(" Hz");
        drumFastKnob.setTextValueSuffix(" Hz");
        rampUpKnob.setTextValueSuffix(" s");
        rampDownKnob.setTextValueSuffix(" s");
        
        speedControlGroup.addAndMakeVisible(hornSlowKnob);
        speedControlGroup.addAndMakeVisible(hornFastKnob);
        speedControlGroup.addAndMakeVisible(drumSlowKnob);
        speedControlGroup.addAndMakeVisible(drumFastKnob);
        speedControlGroup.addAndMakeVisible(rampUpKnob);
        speedControlGroup.addAndMakeVisible(rampDownKnob);
        speedControlGroup.addAndMakeVisible(speedControlCombo);
        speedControlGroup.addAndMakeVisible(brakeButton);
        
        // Labels
        auto addLabel = [&](juce::Label& label, const juce::String& text, juce::Component& parent) {
            label.setText(text, juce::dontSendNotification);
            label.setColour(juce::Label::textColourId, juce::Colours::white);
            label.setJustificationType(juce::Justification::centred);
            parent.addAndMakeVisible(label);
        };
        
        addLabel(hornSlowLabel, "Horn Slow", speedControlGroup);
        addLabel(hornFastLabel, "Horn Fast", speedControlGroup);
        addLabel(drumSlowLabel, "Drum Slow", speedControlGroup);
        addLabel(drumFastLabel, "Drum Fast", speedControlGroup);
        addLabel(rampUpLabel, "Ramp Up", speedControlGroup);
        addLabel(rampDownLabel, "Ramp Down", speedControlGroup);
        
        // Speed control combo
        speedControlCombo.addItem("Slow", 1);
        speedControlCombo.addItem("Fast", 2);
        speedControlCombo.addItem("Brake", 3);
        speedControlCombo.setSelectedItemIndex(0);
        speedControlCombo.setColour(juce::ComboBox::backgroundColourId, juce::Colour(0xff34495E));
        speedControlCombo.setColour(juce::ComboBox::textColourId, juce::Colours::white);
        
        // Brake button
        brakeButton.setButtonText("BRAKE");
        brakeButton.setColour(juce::TextButton::buttonColourId, juce::Colour(0xff8E44AD));
        
        // Tone Control Group
        toneControlGroup.setText("TONE & BALANCE");
        toneControlGroup.setColour(juce::GroupComponent::outlineColourId, juce::Colour(0xff27AE60));
        toneControlGroup.setColour(juce::GroupComponent::textColourId, juce::Colours::white);
        
        driveKnob.setTextValueSuffix(" dB");
        balanceKnob.setTextValueSuffix("%");
        dopplerKnob.setTextValueSuffix("%");
        amDepthKnob.setTextValueSuffix("%");
        stereoSpreadKnob.setTextValueSuffix("%");
        
        toneControlGroup.addAndMakeVisible(driveKnob);
        toneControlGroup.addAndMakeVisible(balanceKnob);
        toneControlGroup.addAndMakeVisible(dopplerKnob);
        toneControlGroup.addAndMakeVisible(amDepthKnob);
        toneControlGroup.addAndMakeVisible(stereoSpreadKnob);
        
        addLabel(driveLabel, "Drive", toneControlGroup);
        addLabel(balanceLabel, "Horn/Drum", toneControlGroup);
        addLabel(dopplerLabel, "Doppler", toneControlGroup);
        addLabel(amDepthLabel, "AM Depth", toneControlGroup);
        addLabel(stereoSpreadLabel, "Stereo Spread", toneControlGroup);
        
        // Output Control Group
        outputControlGroup.setText("OUTPUT & ROUTING");
        outputControlGroup.setColour(juce::GroupComponent::outlineColourId, juce::Colour(0xffF39C12));
        outputControlGroup.setColour(juce::GroupComponent::textColourId, juce::Colours::white);
        
        outputLevelKnob.setTextValueSuffix(" dB");
        
        outputControlGroup.addAndMakeVisible(outputLevelKnob);
        outputControlGroup.addAndMakeVisible(outputModeCombo);
        
        addLabel(outputLevelLabel, "Output Level", outputControlGroup);
        addLabel(outputModeLabel, "Output Mode", outputControlGroup);
        
        outputModeCombo.addItem("Stereo", 1);
        outputModeCombo.addItem("Mono", 2);
        outputModeCombo.addItem("Wet L-Dry R", 3);
        outputModeCombo.addItem("Dual Mono", 4);
        outputModeCombo.setSelectedItemIndex(0);
        outputModeCombo.setColour(juce::ComboBox::backgroundColourId, juce::Colour(0xff34495E));
        outputModeCombo.setColour(juce::ComboBox::textColourId, juce::Colours::white);
    }
    
    void setupLayout()
    {
        // Speed control layout
        auto layoutSpeedControls = [&]() {
            auto bounds = speedControlGroup.getLocalBounds().reduced(15);
            bounds.removeFromTop(20); // Group label space
            
            auto topRow = bounds.removeFromTop(100);
            auto hornSlowBounds = topRow.removeFromLeft(topRow.getWidth() / 2).reduced(5);
            auto hornFastBounds = topRow.reduced(5);
            
            hornSlowLabel.setBounds(hornSlowBounds.removeFromTop(20));
            hornSlowKnob.setBounds(hornSlowBounds);
            hornFastLabel.setBounds(hornFastBounds.removeFromTop(20));
            hornFastKnob.setBounds(hornFastBounds);
            
            auto midRow = bounds.removeFromTop(100);
            auto drumSlowBounds = midRow.removeFromLeft(midRow.getWidth() / 2).reduced(5);
            auto drumFastBounds = midRow.reduced(5);
            
            drumSlowLabel.setBounds(drumSlowBounds.removeFromTop(20));
            drumSlowKnob.setBounds(drumSlowBounds);
            drumFastLabel.setBounds(drumFastBounds.removeFromTop(20));
            drumFastKnob.setBounds(drumFastBounds);
            
            auto bottomRow = bounds.removeFromTop(100);
            auto rampUpBounds = bottomRow.removeFromLeft(bottomRow.getWidth() / 2).reduced(5);
            auto rampDownBounds = bottomRow.reduced(5);
            
            rampUpLabel.setBounds(rampUpBounds.removeFromTop(20));
            rampUpKnob.setBounds(rampUpBounds);
            rampDownLabel.setBounds(rampDownBounds.removeFromTop(20));
            rampDownKnob.setBounds(rampDownBounds);
            
            speedControlCombo.setBounds(bounds.removeFromTop(30).reduced(5));
            brakeButton.setBounds(bounds.removeFromTop(40).reduced(5));
        };
        
        speedControlGroup.resized = layoutSpeedControls;
        
        // Similar layout functions for other groups...
        // (Abbreviated for space - would implement full layout)
    }
    
    void attachSliders()
    {
        using SliderAttachment = juce::AudioProcessorValueTreeState::SliderAttachment;
        using ComboAttachment = juce::AudioProcessorValueTreeState::ComboBoxAttachment;
        using ButtonAttachment = juce::AudioProcessorValueTreeState::ButtonAttachment;
        
        hornSlowAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::HORN_SPEED_SLOW, hornSlowKnob);
        hornFastAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::HORN_SPEED_FAST, hornFastKnob);
        drumSlowAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::DRUM_SPEED_SLOW, drumSlowKnob);
        drumFastAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::DRUM_SPEED_FAST, drumFastKnob);
        rampUpAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::RAMP_TIME_UP, rampUpKnob);
        rampDownAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::RAMP_TIME_DOWN, rampDownKnob);
        
        driveAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::DRIVE_GAIN, driveKnob);
        balanceAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::HORN_DRUM_BALANCE, balanceKnob);
        dopplerAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::DOPPLER_DEPTH, dopplerKnob);
        amDepthAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::AM_DEPTH, amDepthKnob);
        stereoSpreadAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::STEREO_SPREAD, stereoSpreadKnob);
        outputLevelAttachment = std::make_unique<SliderAttachment>(processor.getValueTreeState(), RotaryParams::OUTPUT_LEVEL, outputLevelKnob);
        
        speedControlAttachment = std::make_unique<ComboAttachment>(processor.getValueTreeState(), RotaryParams::SPEED_CONTROL, speedControlCombo);
        outputModeAttachment = std::make_unique<ComboAttachment>(processor.getValueTreeState(), RotaryParams::OUTPUT_MODE, outputModeCombo);
        brakeAttachment = std::make_unique<ButtonAttachment>(processor.getValueTreeState(), RotaryParams::BRAKE_ENGAGE, brakeButton);
    }
    
    LeslieRotaryProcessor& processor;
    
    // Visual components
    RotorVisualizer rotorVisualizer;
    
    // Control groups
    juce::GroupComponent speedControlGroup;
    juce::GroupComponent toneControlGroup;
    juce::GroupComponent outputControlGroup;
    
    // Speed controls
    RotaryKnob hornSlowKnob, hornFastKnob, drumSlowKnob, drumFastKnob;
    RotaryKnob rampUpKnob, rampDownKnob;
    juce::ComboBox speedControlCombo;
    juce::TextButton brakeButton;
    
    // Tone controls
    RotaryKnob driveKnob, balanceKnob, dopplerKnob, amDepthKnob, stereoSpreadKnob;
    
    // Output controls
    RotaryKnob outputLevelKnob;
    juce::ComboBox outputModeCombo;
    
    // Labels
    juce::Label hornSlowLabel, hornFastLabel, drumSlowLabel, drumFastLabel;
    juce::Label rampUpLabel, rampDownLabel;
    juce::Label driveLabel, balanceLabel, dopplerLabel, amDepthLabel, stereoSpreadLabel;
    juce::Label outputLevelLabel, outputModeLabel;
    
    // Attachments
    std::unique_ptr<juce::AudioProcessorValueTreeState::SliderAttachment> hornSlowAttachment, hornFastAttachment;
    std::unique_ptr<juce::AudioProcessorValueTreeState::SliderAttachment> drumSlowAttachment, drumFastAttachment;
    std::unique_ptr<juce::AudioProcessorValueTreeState::SliderAttachment> rampUpAttachment, rampDownAttachment;
    std::unique_ptr<juce::AudioProcessorValueTreeState::SliderAttachment> driveAttachment, balanceAttachment;
    std::unique_ptr<juce::AudioProcessorValueTreeState::SliderAttachment> dopplerAttachment, amDepthAttachment;
    std::unique_ptr<juce::AudioProcessorValueTreeState::SliderAttachment> stereoSpreadAttachment, outputLevelAttachment;
    std::unique_ptr<juce::AudioProcessorValueTreeState::ComboBoxAttachment> speedControlAttachment, outputModeAttachment;
    std::unique_ptr<juce::AudioProcessorValueTreeState::ButtonAttachment> brakeAttachment;
    
    JUCE_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR(LeslieRotaryEditor)
};

//==============================================================================
// Updated processor to use custom editor
//==============================================================================
juce::AudioProcessorEditor* LeslieRotaryProcessor::createEditor()
{
    return new LeslieRotaryEditor(*this);
}
